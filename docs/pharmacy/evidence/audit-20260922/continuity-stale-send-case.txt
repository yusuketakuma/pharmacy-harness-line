it('reproduces a stale ended reminder reaching the provider boundary', async () => {
  const { sqlite, db } = setup();
  try {
    sqlite.exec("UPDATE tenants SET outbound_messaging_paused_at=NULL; UPDATE pharmacy_patients SET relationship='other' WHERE id='patient-a'; UPDATE pharmacy_patients SET relationship='self' WHERE id='patient-000'; UPDATE pharmacy_account_capabilities SET beta_enabled=0;");
    readCredential.mockResolvedValue('synthetic-token');
    const [reminder] = await claimDueNextIntakeExpectations(db, NOW, 1);
    const ended = await endNextIntakeExpectation(db, {
      lineAccountId: reminder.line_account_id, expectationId: reminder.id,
      expectedVersion: reminder.version, staffId: 'synthetic-staff',
      idempotencyKey: 'synthetic-end-request', now: NOW,
    });
    expect(ended.status).toBe('ended');
    const result = await deliverContinuityReminder(reminder, {
      db, proxyBaseUrl: 'https://synthetic.invalid', lineCredentialKey: 'synthetic',
    });
    expect(push).toHaveBeenCalledOnce();
    expect(result).toBe('failed');
    const final = sqlite.prepare("SELECT status,version FROM pharmacy_next_intake_expectations WHERE id='expectation-000'").get();
    expect(final).toEqual({status:'ended',version:2});
    expect(sqlite.prepare('PRAGMA foreign_key_check').all()).toEqual([]);
    writeFileSync('/Users/yusuke/workspace/pharmacy-harness-line/docs/pharmacy/evidence/audit-20260922/continuity-stale-send-result.json', JSON.stringify({result:'REPRODUCED',providerBoundaryCalls:push.mock.calls.length,deliveryResult:result,final,externalCalls:0},null,2));
  } finally {sqlite.close();}
});
