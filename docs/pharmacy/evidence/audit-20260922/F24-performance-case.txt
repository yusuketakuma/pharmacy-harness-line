beforeEach(() => {
  send.mockReset().mockResolvedValue('paused');
  readCredential.mockReset().mockResolvedValue('synthetic-token');
});

function process(db: D1Database, now = NOW, limit = 50) {
  return processDuePrescriptionValidityReminders(db, {
    proxyBaseUrl:'https://synthetic.invalid',lineCredentialKey:'synthetic',now,limit,
  });
}

it.each(['paused', 'failure', 'credential_missing'])(
  'advances past fifty %s rows while preserving validity and claim release', async mode => {
    const { sqlite, db } = setup();
    try {
      if (mode === 'failure') send.mockRejectedValue(new Error('synthetic failure'));
      if (mode === 'credential_missing') readCredential.mockResolvedValue(null);
      const first = await process(db);
      const checked = () => sqlite.prepare('SELECT COUNT(*) AS count FROM pharmacy_prescription_validities WHERE notification_checked_at IS NOT NULL').get();
      expect(checked()).toEqual({count:50});
      const second = await process(db, new Date(NOW.getTime()+21600000));
      expect(checked()).toEqual({count:51});
      expect(first).toEqual({sent:0,failed:mode==='failure'?50:0,skipped:mode==='failure'?0:50,expiredReviewRequired:0});
      expect(second).toEqual(first);
      if (mode === 'credential_missing') expect(send).not.toHaveBeenCalled();
      else expect(send.mock.calls.some(([input])=>input.lineAccountId==='account-b')).toBe(true);
      expect(sqlite.prepare("SELECT COUNT(*) AS count FROM pharmacy_prescription_validities WHERE verification_status='verified' AND valid_until='2026-09-24' AND reminder_due_at='2026-09-21T00:00:00.000Z' AND reminder_claimed_at IS NULL AND reminder_sent_at IS NULL").get()).toEqual({count:51});
      expect(sqlite.prepare('PRAGMA foreign_key_check').all()).toEqual([]);
    } finally {sqlite.close();}
  },
);

it('retains the previous-schema delivery and release contract', async () => {
  const { sqlite, db } = setup(true);
  try {
    expect(await process(db)).toEqual({sent:0,failed:0,skipped:50,expiredReviewRequired:0});
    await process(db,new Date(NOW.getTime()+21600000));
    expect(send).toHaveBeenCalledTimes(100);
    expect(send.mock.calls.every(([input])=>input.lineAccountId==='account-a')).toBe(true);
    expect(sqlite.prepare('PRAGMA table_info(pharmacy_prescription_validities)').all().some(row=>(row as {name:string}).name==='notification_checked_at')).toBe(false);
  } finally {sqlite.close();}
});

it('retries checked rows after resumption and stamps only confirmed delivery', async () => {
  const { sqlite, db } = setup();
  try {
    await process(db);
    await process(db,new Date(NOW.getTime()+21600000));
    sqlite.exec("UPDATE tenants SET outbound_messaging_paused_at=NULL WHERE id='tenant-a'");
    send.mockResolvedValue('sent');
    const result=await process(db,new Date(NOW.getTime()+43200000));
    expect(result.sent).toBe(50);
    expect(sqlite.prepare('SELECT COUNT(*) AS count FROM pharmacy_prescription_validities WHERE reminder_sent_at IS NOT NULL AND reminder_claimed_at IS NULL').get()).toEqual({count:50});
  } finally {sqlite.close();}
});

it.each(['tenant', 'account', 'claim'])(
  'does not advance or send a row after a %s conflict', async conflict => {
    const { sqlite, db } = setup();
    const guarded={...db,prepare:(sql:string)=>{
      if(sql.includes('SET reminder_claimed_at = ?')) {
        if(conflict==='tenant') sqlite.exec("UPDATE tenants SET status='suspended' WHERE id='tenant-a'");
        if(conflict==='account') sqlite.exec("UPDATE line_accounts SET is_active=0 WHERE id='account-a'");
        if(conflict==='claim') sqlite.prepare("UPDATE pharmacy_prescription_validities SET reminder_claimed_at=? WHERE submission_id='sub-000'").run(NOW.toISOString());
      }
      return db.prepare(sql);
    }} as D1Database;
    try {
      expect(await process(guarded,NOW,1)).toEqual({sent:0,failed:0,skipped:1,expiredReviewRequired:0});
      expect(send).not.toHaveBeenCalled();
      expect(sqlite.prepare("SELECT notification_checked_at FROM pharmacy_prescription_validities WHERE submission_id='sub-000'").get()).toEqual({notification_checked_at:null});
    } finally {sqlite.close();}
  },
);

it('does not move scheduling metadata backwards', async () => {
  const { sqlite, db } = setup();
  try {
    const later=new Date(NOW.getTime()+21600000);
    await process(db,later,100);
    await process(db,NOW,100);
    expect(sqlite.prepare('SELECT DISTINCT notification_checked_at FROM pharmacy_prescription_validities').all()).toEqual([{notification_checked_at:later.toISOString()}]);
  } finally {sqlite.close();}
});
