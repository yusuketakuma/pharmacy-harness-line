import { writeFileSync } from 'node:fs';
import { afterEach, describe, expect, it, vi } from 'vitest';

const db = vi.hoisted(() => ({
  getActiveTenantLineAccounts: vi.fn(),
  updateLineAccount: vi.fn(),
}));
const credentials = vi.hoisted(() => ({
  readLineCredential: vi.fn(),
  writeLineCredential: vi.fn(),
}));
const accounts = vi.hoisted(() => ({
  updateEncryptedLineAccount: vi.fn(),
}));

vi.mock('@line-crm/db', () => db);
vi.mock('../custom/pharmacy/provisioning/line-credential-store.js', () => credentials);
vi.mock('../custom/pharmacy/provisioning/line-account-store.js', () => accounts);

import { refreshLineAccessTokens } from './token-refresh.js';

it('measures a bounded synthetic provider wait', async () => {
  db.getActiveTenantLineAccounts.mockResolvedValue(['a','b'].map(id => ({
    id, tenant_id: 'tenant-'+id, pharmacy_mode: 1, channel_id: 'channel-'+id,
    is_active: 1, token_expires_at: null, updated_at: '2026-09-22T00:00:00.000Z',
  })));
  credentials.readLineCredential.mockResolvedValue('synthetic-secret');
  accounts.updateEncryptedLineAccount.mockResolvedValue({});
  let release!: (value: Response) => void;
  let reached!: () => void;
  const entered = new Promise<void>(resolve => { reached=resolve; });
  const pending = new Promise<Response>(resolve => { release=resolve; });
  const response = () => new Response(JSON.stringify({ access_token:'synthetic-token', expires_in:3600, token_type:'Bearer' }));
  const fetchMock=vi.fn().mockImplementationOnce(() => { reached(); return pending; }).mockImplementation(async () => response());
  vi.stubGlobal('fetch',fetchMock);
  let settled=false;
  const start=performance.now();
  const running=refreshLineAccessTokens({} as D1Database,{lineCredentialKey:'synthetic-root'}).then(()=>{settled=true;});
  await entered;
  await new Promise(resolve=>setTimeout(resolve,100));
  const observed={heldMs:performance.now()-start,settled,requests:fetchMock.mock.calls.length,writes:accounts.updateEncryptedLineAccount.mock.calls.length,signalPresent:Boolean(fetchMock.mock.calls[0]?.[1]?.signal)};
  release(response());
  await running;
  expect(observed.settled).toBe(false);
  expect(observed.requests).toBe(1);
  expect(observed.writes).toBe(0);
  expect(fetchMock).toHaveBeenCalledTimes(2);
  expect(accounts.updateEncryptedLineAccount).toHaveBeenCalledTimes(2);
  writeFileSync('/Users/yusuke/workspace/pharmacy-harness-line/docs/pharmacy/evidence/audit-20260922/token-refresh-delay-result.json',JSON.stringify({observed,settledAfterRelease:settled,totalMs:performance.now()-start,requestsAfterRelease:fetchMock.mock.calls.length,externalRequestsSent:0},null,2));
  vi.unstubAllGlobals();
});
