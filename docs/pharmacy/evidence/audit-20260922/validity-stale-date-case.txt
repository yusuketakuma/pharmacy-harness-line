it('reproduces a stale validity date being sent after staff correction', async () => {
  const {savePrescriptionValidity} = await import('./repository.js');
  const {sqlite,db}=setup();
  try {
    sqlite.exec("UPDATE tenants SET outbound_messaging_paused_at=NULL; UPDATE pharmacy_account_capabilities SET beta_enabled=0");
    readCredential.mockImplementationOnce(async()=>{
      await savePrescriptionValidity(db,{
        lineAccountId:'account-a',submissionId:'sub-000',issuedOn:'2026-09-22',validUntil:'2026-09-25',
        validityBasis:'default_4_days',verificationStatus:'verified',staffId:'synthetic-staff',
      });
      return 'synthetic-token';
    });
    const result=await processDuePrescriptionValidityReminders(db,{proxyBaseUrl:'https://synthetic.invalid',lineCredentialKey:'synthetic',now:NOW});
    expect(push).toHaveBeenCalledOnce();
    expect(JSON.stringify(push.mock.calls[0][3])).toContain('2026-09-24');
    const saved=sqlite.prepare("SELECT valid_until,reminder_sent_at,reminder_claimed_at FROM pharmacy_prescription_validities WHERE submission_id='sub-000'").get();
    expect(saved).toEqual({valid_until:'2026-09-25',reminder_sent_at:null,reminder_claimed_at:null});
    expect(result.sent).toBe(1);
    expect(sqlite.prepare('PRAGMA foreign_key_check').all()).toEqual([]);
    writeFileSync('/Users/yusuke/workspace/pharmacy-harness-line/docs/pharmacy/evidence/audit-20260922/validity-stale-date-result.json',JSON.stringify({result:'REPRODUCED',processor:result,saved,providerCalls:push.mock.calls.length,externalCalls:0},null,2));
  } finally {sqlite.close();}
});
